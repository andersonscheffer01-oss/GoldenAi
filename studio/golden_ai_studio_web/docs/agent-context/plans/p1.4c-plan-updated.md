## ADJUSTMENT 1 — TURN MANAGER (Single Authority)

### Design

```python
# jarvis_core/streaming/turn_manager.py

class TurnManager:
    """Single authority for current turn ID.
    
    Thread-safe. All components check this to detect stale turns.
    """
    
    def __init__(self):
        self._lock = threading.Lock()
        self._current_turn_id = 0
        self._cancel_events: dict[int, threading.Event] = {}
    
    def new_turn(self) -> int:
        """Create a new turn, cancel previous. Returns new turn_id."""
        with self._lock:
            # Cancel previous turn if exists
            if self._current_turn_id in self._cancel_events:
                self._cancel_events[self._current_turn_id].set()
            
            self._current_turn_id += 1
            turn_id = self._current_turn_id
            self._cancel_events[turn_id] = threading.Event()
            return turn_id
    
    def get_cancel_event(self, turn_id: int) -> threading.Event:
        """Get the cancel event for a specific turn."""
        with self._lock:
            return self._cancel_events.get(turn_id)
    
    def cancel_current(self):
        """Cancel the current turn."""
        with self._lock:
            if self._current_turn_id in self._cancel_events:
                self._cancel_events[self._current_turn_id].set()
    
    def is_stale(self, turn_id: int) -> bool:
        """Check if a turn_id is stale (not current)."""
        with self._lock:
            return turn_id != self._current_turn_id
    
    def cleanup_turn(self, turn_id: int):
        """Remove old turn data."""
        with self._lock:
            self._cancel_events.pop(turn_id, None)
    
    @property
    def current_turn_id(self) -> int:
        with self._lock:
            return self._current_turn_id
```

### Integration

```
VoiceService._loop():
  turn_id = turn_manager.new_turn()
  # ... pass turn_id to orchestrator ...
  
  # On barge-in:
  turn_manager.cancel_current()

  cancel_event = turn_manager.get_cancel_event(turn_id)
  # Use threading.Event instead of asyncio.Event

    return 0  # Discard stale chunk
```

## ADJUSTMENT 2 — CROSS-THREAD CANCEL

### Threading Analysis

| Component | Thread | Cancel Source | Mechanism |
|-----------|--------|---------------|-----------|
| StreamingOrchestrator | asyncio.run() on jarvis-voice | Frame loop (async) or VoiceService (sync) | `threading.Event` |
| TTSPipeline | Same event loop as orchestrator | Via cancel_event | `threading.Event` checked in sync code |
| AudioPlayer | Thread pool worker | Via orchestrator | `threading.Event` + `Popen.terminate()` |
| Provider stream | Same event loop | Via cancel_event | Break in async for loop |

### Design: Use `threading.Event` Everywhere

```python
# In TurnManager.new_turn():
self._cancel_events[turn_id] = threading.Event()  # NOT asyncio.Event

# In Orchestrator.run():
cancel_event = turn_manager.get_cancel_event(turn_id)

# Check in sync code (provider loop, TTS):
if cancel_event.is_set():
    break  # or return None

# In AudioPlayer.stop():
cancel_event.set()  # Thread-safe from any thread
```

### Thread-Safe AudioPlayer Stop

```python
class AudioPlayer:
    def __init__(self):
        self._playing = False
        self._current_process: Optional[subprocess.Popen] = None
        self._cancel_event: Optional[threading.Event] = None
    
    async def play(self, chunk, cancel_event=None):
        self._cancel_event = cancel_event
        self._playing = True
        self._current_process = subprocess.Popen(...)
        
        try:
            await asyncio.get_event_loop().run_in_executor(
                None, self._wait_for_process
            )
        finally:
            self._playing = False
            self._current_process = None
    
    def _wait_for_process(self):
        """Wait for process, checking cancel_event periodically."""
        while self._current_process and self._current_process.poll() is None:
            if self._cancel_event and self._cancel_event.is_set():
                self._terminate_process()
                return
            self._current_process.wait(timeout=0.1)
    
    def _terminate_process(self):
        """Terminate PowerShell child process safely."""
        if self._current_process:
            try:
                self._current_process.terminate()
                self._current_process.wait(timeout=2.0)
            except:
                self._current_process.kill()
            self._current_process = None
    
    def stop(self):
        """Immediate stop — called from any thread."""
        self._terminate_process()
        self._playing = False
```

```

    ...
    audio = await tts.synthesize_and_get(text, cancel_event)
```

## ADJUSTMENT 3 — MULTI-LAYER SELF-AUDIO PROTECTION

### Design: 6-Layer Defense

```
Layer 1: STATE GATE
  BargeInController.check_barge_in() → only in SPEAKING state
  Status: EXISTS (realtime_voice/barge_in.py:70)

Layer 2: GRACE PERIOD
  First 0.2s after TTS start → suppress barge-in
  Status: EXISTS (realtime_voice/barge_in.py:75)

Layer 3: TTS EPOCH
  _tts_epoch incremented before TTS → capture during TTS is stale
  Status: EXISTS (voice_service.py:97)

Layer 4: VAD PERSISTENCE
  Require N consecutive speech windows (~96ms) before confirming speech
  Status: EXISTS (realtime_voice/vad.py, speech_start_windows=3)

Layer 5: MINIMUM UTTERANCE
  Audio < 500ms → discard (not real speech)
  Status: EXISTS (realtime_voice/constants.py:MIN_UTTERANCE_MS=500)

Layer 6: ACTIVE PLAYBACK HEURISTIC (NEW)
  During active TTS playback, raise speech threshold:
  - Normal speech threshold: 110
  - During TTS playback: 300
  This means: weak signals (TTS leakage) are ignored,
  but strong signals (user speaking loudly over TTS) still trigger.
```

### Implementation

```python
# In _frame_loop(), during SPEAKING state:
if self._sm.state == ConversationalState.SPEAKING:
    # Layer 1: State gate
    if not self._barge_in.check_barge_in():
        continue
    
    # Layer 6: Active playback threshold
    rms = rms_energy(frame)
    if self._barge_in.tts_active and rms < self._barge_in.active_playback_threshold:
        # Weak signal during TTS — likely leakage, ignore
        continue
    
    # All layers pass → barge-in confirmed
    self._handle_barge_in()
```

### Active Playback Threshold Logic

```python
class BargeInController:
    # During TTS playback, require stronger signal to confirm barge-in
    ACTIVE_PLAYBACK_THRESHOLD = 300  # vs normal 110
    NORMAL_THRESHOLD = 110
    
    def get_speech_threshold(self) -> float:
        """Return threshold based on TTS state."""
        if self._tts_active:
            return self.ACTIVE_PLAYBACK_THRESHOLD
        return self.NORMAL_THRESHOLD
```

### Behavior Matrix

| Scenario | State | Grace | RMS | VAD Windows | Min Duration | Result |
|----------|-------|-------|-----|-------------|--------------|--------|
| TTS leakage (speakers) | SPEAKING | >0.2s | 150-250 | 1-2 | <500ms | IGNORED (Layers 6, 5) |
| User speaks over TTS | SPEAKING | >0.2s | 350+ | 3+ | 500ms+ | BARGE-IN |
| Noise during TTS | SPEAKING | >0.2s | 200 | 1 | <500ms | IGNORED (Layers 4, 5) |
| User after TTS ends | IDLE/LISTENING | N/A | 110+ | 3+ | 500ms+ | LISTENING (no barge-in needed) |
| Brief noise | LISTENING | N/A | 200 | 1 | <500ms | IGNORED (Layers 4, 5) |

## ADJUSTMENT 4 — AUDIOPLAYER REAL STOP

### Design

```python
class AudioPlayer:
    def __init__(self):
        self._playing = False
        self._current_process: Optional[subprocess.Popen] = None
    
    async def play(self, chunk: AudioChunk, 
                   cancel_event: Optional[threading.Event] = None) -> float:
        if not chunk.audio_path or not Path(chunk.audio_path).exists():
            return 0
        
        self._playing = True
        start = time.perf_counter()
        
        try:
            uri = Path(chunk.audio_path).resolve().as_uri()
            ps = (
                "Add-Type -AssemblyName PresentationCore;"
                "$p = New-Object System.Windows.Media.MediaPlayer;"
                f"$p.Open([Uri]'{uri}');"
                "for($i=0;$i -lt 50 -and -not $p.NaturalDuration.HasTimeSpan;$i++)"
                "{Start-Sleep -Milliseconds 100};"
                "if($p.NaturalDuration.HasTimeSpan)"
                "{$d=[math]::Ceiling($p.NaturalDuration.TimeSpan.TotalSeconds+0.5)}"
                "else{$d=8};"
                "$p.Play(); Start-Sleep -Seconds $d; $p.Close()"
            )
            
            self._current_process = subprocess.Popen(
                ["powershell", "-NoProfile", "-Command", ps],
                stdout=subprocess.DEVNULL,
                stderr=subprocess.DEVNULL,
            )
            
            # Wait in executor, checking cancel periodically
            await asyncio.get_event_loop().run_in_executor(
                None, self._wait_for_process, cancel_event
            )
            
        except Exception as e:
            log.warning("Playback failed: %s", e)
        finally:
            self._playing = False
            self._current_process = None
        
        return (time.perf_counter() - start) * 1000
    
    def _wait_for_process(self, cancel_event: Optional[threading.Event] = None):
        """Wait for subprocess, checking cancel_event every 100ms."""
        while self._current_process and self._current_process.poll() is None:
            if cancel_event and cancel_event.is_set():
                self._terminate_process()
                return
            try:
                self._current_process.wait(timeout=0.1)
            except subprocess.TimeoutExpired:
                continue
    
    def _terminate_process(self):
        """Terminate the subprocess and its children safely."""
        if self._current_process is None:
            return
        
        try:
            # Get child processes (PowerShell spawns mediaplayer)
            parent = self._current_process
            
            # Try terminate first
            parent.terminate()
            try:
                parent.wait(timeout=2.0)
            except subprocess.TimeoutExpired:
                parent.kill()
                parent.wait(timeout=1.0)
            
        except Exception:
            pass
        finally:
            self._current_process = None
    
    def stop(self):
        """Immediate stop from any thread."""
        self._terminate_process()
        self._playing = False
```
